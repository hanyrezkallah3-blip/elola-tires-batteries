// ======================================================
// NORMALIZE
// ======================================================

const norm = value =>
  String(value ?? '')
    .trim()
    .toLowerCase()
    .replace(/ط£|ط¥|ط¢/g, 'ط§')
    .replace(/ط©/g, 'ظ‡')
    .replace(/ظ‰/g, 'ظٹ')
    .replace(/ظٹظژ|ظٹظڈ|ظٹظگ|ظژ|ظڈ|ظگ|ظ‘|ظ’/g, '')
    .replace(/\s+/g, '')


// ======================================================
// NORMALIZE TYPE
// ======================================================

const normalizeType = value => {

  const type =
    norm(value)

  if (
    [
      'tire',
      'tires',
      'tyre',
      'tyres',
      'ط§ط·ط§ط±',
      'ط§ط·ط§ط±ط§طھ'
    ].includes(type)
  ) {
    return 'tire'
  }

  if (
    [
      'battery',
      'batteries',
      'ط¨ط·ط§ط±ظٹظ‡',
      'ط¨ط·ط§ط±ظٹط§طھ'
    ].includes(type)
  ) {
    return 'battery'
  }

  if (
    [
      'oil',
      'oils',
      'ط²ظٹطھ',
      'ط²ظٹظˆطھ'
    ].includes(type)
  ) {
    return 'oil'
  }

  return type
}


// ======================================================
// VALUE EQUALITY
// ======================================================

const valuesMatch = (
  actual,
  wanted
) => {

  const left =
    norm(actual)

  const right =
    norm(wanted)

  if (
    !left ||
    !right
  ) {
    return false
  }

  return (
    left === right ||
    left.includes(right) ||
    right.includes(left)
  )
}


// ======================================================
// TIRE SIZE PARSER
// ======================================================

const parseTireSize = value => {

  const input =
    String(value ?? '')
      .trim()
      .replace(/\s+/g, '')
      .replace(/أ—/g, '*')
      .replace(/x/gi, '*')
      .replace(/-/g, '/')

  if (!input) {
    return null
  }

  const threePart =
    input.match(
      /^(\d+(?:\.\d+)?)[\/\\*](\d+(?:\.\d+)?)[\/\\*](\d+(?:\.\d+)?)$/
    )

  if (threePart) {

    return {

      width:
        threePart[1],

      profile:
        threePart[2],

      rim:
        threePart[3],

      format:
        'three-part'

    }
  }

  const twoPart =
    input.match(
      /^(\d+(?:\.\d+)?)[\/\\*](\d+(?:\.\d+)?)$/
    )

  if (twoPart) {

    return {

      width:
        twoPart[1],

      profile:
        '',

      rim:
        twoPart[2],

      format:
        'two-part'

    }
  }

  return null
}


// ======================================================
// FLATTEN SEARCH RESULTS
// ======================================================

const flattenSearchResults = (
  tab,
  data
) => {

  if (
    Array.isArray(data)
  ) {
    return data
  }

  if (
    !data ||
    typeof data !== 'object'
  ) {
    return []
  }

  if (
    tab === 'vehicle'
  ) {

    return [

      ...(Array.isArray(data.tires)
        ? data.tires
        : []),

      ...(Array.isArray(data.batteries)
        ? data.batteries
        : []),

      ...(Array.isArray(data.oils)
        ? data.oils
        : []),

      ...(Array.isArray(data.parts)
        ? data.parts
        : []),

      ...(Array.isArray(data.products)
        ? data.products
        : [])

    ]
  }

  if (
    tab === 'tire'
  ) {

    return Array.isArray(data.tires)
      ? data.tires
      : []
  }

  if (
    tab === 'battery'
  ) {

    return Array.isArray(data.batteries)
      ? data.batteries
      : []
  }

  if (
    tab === 'oil'
  ) {

    return Array.isArray(data.oils)
      ? data.oils
      : []
  }

  return []
}


// ======================================================
// IDS
// ======================================================

const idsEqual = (
  a,
  b
) => {

  const left = [

    a?.productId,
    a?.id,
    a?.sku,
    a?.barcode

  ]
    .filter(Boolean)
    .map(norm)

  const right = [

    b?.productId,
    b?.id,
    b?.sku,
    b?.barcode

  ]
    .filter(Boolean)
    .map(norm)

  return left.some(
    id =>
      right.includes(id)
  )
}

const activeOfferFor = product => {

  const websiteState =

  const websiteProducts =

      websiteState?.products
    )
      ? websiteState.products
      : []

  const websiteProduct =
    websiteProducts.find(
      item =>

        )
    )

  if (
    !websiteProduct
  ) {

    return {

      websiteProduct:
        null,

      offer:
        null

    }
  }

  const offers =
    Array.isArray(
      websiteState?.offers
    )
      ? websiteState.offers
      : []

  const now =
    new Date()

  const offer =
    offers.find(
      item => {

        if (
          !item ||
          item.active === false ||
          item.productId == null
        ) {
          return false
        }

        if (
          !idsEqual(
            websiteProduct,
            {
              productId:
                item.productId
            }
          )
        ) {
          return false
        }

        if (
          item.startDate
        ) {

          const date =
            new Date(
              item.startDate
            )

          if (
            !Number.isNaN(
              date.getTime()
            ) &&
            now < date
          ) {
            return false
          }
        }

        if (
          item.endDate
        ) {

// COLLECT VALUES
// ======================================================

const collectValues = (
  product,
  fields
) => {

  const values = []

  const add = value => {

    if (
      value == null
    ) {
      return
    }

    if (
      Array.isArray(value)
    ) {

      value.forEach(
        item =>
          add(item)
      )

      return
    }

    if (
      typeof value === 'object'
    ) {

      Object.values(
        value
      ).forEach(
        item =>
          add(item)
      )

      return
    }

    values.push(
      String(value)
    )
  }

  const specifications =
    product?.specifications ||
    {}

  const attributes =
    product?.attributes ||
    {}

  const battery =
    product?.battery ||
    {}

  const oil =
    product?.oil ||
    {}

  fields.forEach(
    field => {

      add(
        product?.[field]
      )

      add(
        battery?.[field]
      )

      add(
        oil?.[field]
      )

      add(
        specifications?.[field]
      )

      add(
        attributes?.[field]
      )

      add(
        specifications?.battery?.[field]
      )

      add(
        specifications?.oil?.[field]
      )

      add(
        attributes?.battery?.[field]
      )

      add(
        attributes?.oil?.[field]
      )
    }
  )

  return values
}

const extractOilGrades = value => {

  const text =
    normalizeOilValue(
      value
    )

  if (
    !text
  ) {
    return []
  }

  const grades = []

  const rangeMatches =
    text.match(
      /\b\d{1,3}w-?\d{1,3}\b/gi
    )

  if (
    Array.isArray(
      rangeMatches
    )
  ) {

    rangeMatches.forEach(
      item => {

        grades.push(
          norm(item)
        )
      }
    )
  }

  const wMatches =
    text.match(
      /\b\d{1,3}w\b/gi
    )

  if (
    Array.isArray(
      wMatches
    )
  ) {

    wMatches.forEach(
      item =>
        grades.push(
          norm(item)
        )
    )
  }

  const numberMatches =
    text.match(
      /\b\d{1,3}\b/g
    )

  if (
    Array.isArray(
      numberMatches
    )
  ) {

    numberMatches.forEach(
      item =>
        grades.push(
          norm(item)
        )
    )
  }

  return [

    ...new Set(
      grades.filter(Boolean)
    )

  ]
}

// ======================================================

const oilMatches = (
  product,
  viscosity
) => {

  const wanted =
    normalizeOilValue(
      viscosity
    )

  if (
    !wanted
  ) {
    return false
  }

  const values =
    collectValues(
      product,
      [
        'viscosity',
        'viscosityGrade',
        'grade',
        'oilGrade',
        'oilViscosity',
        'sae',
        'SAE',
        'weight',
        'oilWeight',
        'specification',
        'description',
        'name',
        'productName',
        'title',
        'code',
        'sku'
      ]
    )

  const wantedGrades =
    extractOilGrades(
      wanted
    )

  return values.some(
    value => {

      const actual =
        normalizeOilValue(
          value
        )

      if (
        !actual
      ) {
        return false
      }

      if (
        actual === wanted
      ) {
        return true
      }

      if (
        actual.includes(
          wanted
        )
      ) {
        return true
      }

      const actualGrades =
        extractOilGrades(
          actual
        )

      return wantedGrades.some(
        wantedGrade =>
          actualGrades.some(
            actualGrade => {

              if (
                actualGrade ===
                wantedGrade
              ) {
                return true
              }

              if (
                /^\d+$/.test(
                  wantedGrade
                ) &&
                /^\d+w$/.test(
                  actualGrade
                )
              ) {

                return (
                  actualGrade.slice(
                    0,
                    -1
                  ) ===
                  wantedGrade
                )
              }

              if (
                /^\d+w$/.test(
                  wantedGrade
                ) &&
                actualGrade.startsWith(
                  wantedGrade
                )
              ) {
                return true
              }

              return false
            }
          )
      )
    }
  )
}

                existing?.stock ??
                0
              )
            ),

          salePrice:
            Number(
              product?.salePrice ??
              product?.sellingPrice ??
              product?.price ??
              existing?.salePrice ??
              existing?.sellingPrice ??
              existing?.price ??
              0
            )

        }
      )
    }
  )

  return Array.from(
    map.values()
  )
}

const mergeWarehouseSearchResults = (
  tab,
  controllerResults,
  form,
  parsed
) => {

  const warehouse =
    warehouseProducts()

  const catalog =
    websiteProducts()

  let warehouseMatches =
    []

  let catalogMatches =
    []

  if (
    tab === 'tire'
  ) {

    warehouseMatches =
      warehouse.filter(
        product =>
          productType(
            product
          ) === 'tire' &&
          tireMatches(
            product,
            parsed
          )
      )

    catalogMatches =
      catalog.filter(
        product =>
          productType(
            product
          ) === 'tire' &&
          tireMatches(
            product,
            parsed
          )
      )
  }


  if (
    tab === 'battery'
  ) {

    warehouseMatches =
      warehouse.filter(
        product => {

          const type =
            productType(
              product
            )

          return (
            type === 'battery' &&
            batteryMatches(
              product,
              form.capacity
            )
          )
        }
      )

    catalogMatches =
      catalog.filter(
        product => {

          const type =
            productType(
              product
            )

          return (
            type === 'battery' &&
            batteryMatches(
              product,
              form.capacity
            )
          )
        }
      )
  }


  if (
    tab === 'oil'
  ) {

    warehouseMatches =
      warehouse.filter(
        product => {

          const type =
            productType(
              product
            )

          return (
            type === 'oil' &&
            oilMatches(
              product,
              form.viscosity
            )
          )
        }
      )

    catalogMatches =
      catalog.filter(
        product => {

          const type =
            productType(
              product
            )

          return (
            type === 'oil' &&
            oilMatches(
              product,
              form.viscosity
            )
          )
        }
      )
  }


  if (
    tab === 'vehicle'
  ) {

    warehouseMatches =
      warehouse.filter(
        product =>
          vehicleMatches(
            product,
            form
          )
      )

    catalogMatches =
      catalog.filter(
        product =>
          vehicleMatches(
            product,
            form
          )
      )
  }


  const normalizedWarehouse =
    warehouseMatches.map(
      publicWarehouseProduct
    )

  const normalizedCatalog =
    catalogMatches.map(
      publicWarehouseProduct
    )

  const normalizedController =
    Array.isArray(
      controllerResults
    )
      ? controllerResults.map(
          publicWarehouseProduct
        )
      : []

  return mergeProducts(
    [

      ...normalizedWarehouse,

      ...normalizedCatalog,

      ...normalizedController

    ]
  )
}


// ======================================================
// AI VEHICLE SEARCH
//
// This is the important new path.
//
// Example:
//
// "طھظˆظٹظˆطھط§ ظƒط±ظˆظ„ط§ 2020"
//
// The text is sent directly to VehicleEngine,
// which resolves the vehicle using VehicleAIEngine,
// then performs OEM compatibility and product matching.
// ======================================================

const searchVehicleWithAI = async query => {

  const text =
    String(
      query ?? ''
    ).trim()

  if (
    !text
  ) {
    return {

      response:
        null,

      results:
        []

    }
  }


  const products =
    allSearchableProducts()


  try {

    console.log(
      '[Vehicle AI] Searching:',
      text
    )


    // --------------------------------------------------
    // First try the AI Vehicle Engine.
    //
    // VehicleEngine itself calls:
    //
    // VehicleAIEngine
    // VehicleProvider
    // OEMCompatibilityEngine
    // VehicleCompatibilityEngine
    // --------------------------------------------------

    const response =
      await VehicleEngine.search({

        vehicleType:
          'car',

        make:
          '',

        model:
          '',

        year:
          '',

        query:
          text,

        products

      })


    console.log(
      '[Vehicle AI] Engine response:',
      response
    )


    const engineProducts =
      flattenSearchResults(
        'vehicle',
        response
      )


    if (
      engineProducts.length > 0
    ) {

      return {

        response,

        results:
          mergeProducts(
            engineProducts.map(
              publicWarehouseProduct
            )
          )

      }
    }


    // --------------------------------------------------
    // Fallback:
    //
    // Some VehicleEngine versions resolve the vehicle
    // only from make/model/year arguments.
    //
    // Ask the controller using the free text as a
    // secondary path if supported.
    // --------------------------------------------------

    try {

      const controllerResponse =
        await VehicleSearchController.searchVehicle({

          query:
            text,

          vehicleType:
            'car',

          make:
            '',

          model:
            '',

          year:
            ''

        })


      const controllerProducts =
        flattenSearchResults(
          'vehicle',
          controllerResponse
        )


      if (
        controllerProducts.length > 0
      ) {

        return {

          response:
            controllerResponse,

          results:
            mergeProducts(
              controllerProducts.map(
                publicWarehouseProduct
              )
            )

        }
      }

    }
    catch (
      controllerError
    ) {

      console.warn(
        '[Vehicle AI] Controller free-text search failed:',
        controllerError
      )
    }


    return {

      response,

      results:
        []

    }

  }
  catch (
    error
  ) {

    console.error(
      '[Vehicle AI] Search failed:',
      error
    )

    return {

      response:
        null,

      results:
        []

    }
  }
}

const normalizeText = value => {

  return String(
    value ?? ''
  )
    .trim()
    .toLowerCase()
    .replace(/ط£|ط¥|ط¢/g, 'ط§')
    .replace(/ط©/g, 'ظ‡')
    .replace(/ظ‰/g, 'ظٹ')
    .replace(/[\u064B-\u065F\u0670]/g, '')

}

const normalizeType = value => {

    normalizeText(value)

    ].includes(type)
  ) {

    return 'oil'

  }

}

// COLLECT SIZE VALUES

const collectSizes = value => {

  const sizes = []


  const add = item => {

    if (
      item === null ||
      item === undefined ||
      item === ''
    ) {

      return

    }


    if (
      Array.isArray(item)
    ) {

      item.forEach(add)

      return

    }


    if (
      typeof item === 'string'
    ) {

      item
        .split(',')
        .forEach(
          part => {

            const size =
              String(part)
                .trim()

            if (size) {

              sizes.push(size)

            }

          }
        )

      return

    }


    if (
      typeof item === 'object'
    ) {

      Object.values(item)
        .forEach(add)

      return

    }


    sizes.push(
      String(item).trim()
    )

  }


  add(value)


  return sizes

}


// ======================================================

const uniqueSizes = sizes => {

  const seen =
    new Set()


  return sizes.filter(
    size => {

      const normalized =
        normalizeText(size)


      if (!normalized) {

        return false

      }


      if (
        seen.has(normalized)
      ) {

        return false

      }


      seen.add(normalized)

      return true

    }
  )

}

            item.alternateSizes

        model,

        year ?? null,

const tireMatchesOEM = (
  product,
  oem

  if (!oem) {

    return false

  }


  const productTire =
    extractProductTire(
      product
    )


  const oemValues =
    collectOEMTireValues(
      oem
    )


  const directArrays =
    getOEMTireSizeArrays(
      oem
    )


  const directSizes =
    collectValues(
      directArrays
    )


  const allOEMValues = [

    ...oemValues,

    ...directSizes

  ]


  if (
    allOEMValues.length === 0
  ) {

    return false

  }


  // ----------------------------------------------------
  // NUMERIC SIZE MATCH
  // ----------------------------------------------------

  for (

    const value
    of allOEMValues

  ) {

    const parsed =
      parseTireSize(
        value
      )


    if (!parsed) {

      continue

    }


    // --------------------------------------------------
    // THREE-PART SIZE
    //
    // 205/55R16
    // --------------------------------------------------

    if (

      productTire.width !== null &&

      productTire.profile !== null &&

      productTire.rim !== null &&

      parsed.width !== null &&

      parsed.profile !== null &&

      parsed.rim !== null &&

      productTire.width ===
        parsed.width &&

      productTire.profile ===
        parsed.profile &&

      productTire.rim ===
        parsed.rim

    ) {

      return true

    }


    // --------------------------------------------------
    // TWO-PART SIZE
    //
    // 1200/24
    // --------------------------------------------------

    if (

      productTire.width !== null &&

      productTire.rim !== null &&

      parsed.width !== null &&

      parsed.rim !== null &&

      parsed.profile === null &&

      productTire.width ===
        parsed.width &&

      productTire.rim ===
        parsed.rim

    ) {

      return true

    }


    // --------------------------------------------------
    // TEXT SIZE
    // --------------------------------------------------

    if (

      productTire.size &&

      valuesMatch(

        productTire.size,

        value

      )

    ) {

      return true

    }

  }


  // ----------------------------------------------------
  // FALLBACK PRODUCT TEXT
  // ----------------------------------------------------

  const productValues =

    collectValues([

      productTire.size,

      product?.tireSize,

      product?.size,

      product?.dimension,

      product?.dimensions,

      product?.width,

      product?.profile,

      product?.rim,

      product?.name,

      product?.productName,

      product?.title,

      product?.description,

      product?.sku

    ])


  return allOEMValues.some(

    wanted =>

      productValues.some(

        actual =>

          valuesMatch(

            actual,

            wanted

          )

      )

  )

}

  model,

  year,

  vehicleType,

  oem

}) => {

  if (!product) {

    return false

  }


  // ----------------------------------------------------
  // FIRST: EXPLICIT VEHICLE COMPATIBILITY
  // ----------------------------------------------------

  if (

    explicitVehicleMatch({

      product,

      make,

      model,

      year,

      vehicleType

    })

  ) {

    return true

  }


  // ----------------------------------------------------
  // SECOND: OEM TECHNICAL COMPATIBILITY
  // ----------------------------------------------------

  const type =

    normalizeProductType(
      product
    )


  if (
    type === 'tire'
  ) {

    return tireMatchesOEM(

      product,

      oem

    )

  }


  if (
    type === 'battery'
  ) {

    return batteryMatchesOEM(

      product,

      oem

    )

  }


  if (
    type === 'oil'
  ) {

    return oilMatchesOEM(

      product,

      oem

    )

  }


  return false

}

export class VehicleEngine {


  // ====================================================
  // FIND VEHICLE
  // ====================================================

  static async findVehicle({

    make,

    model,

    year,

    vehicleType

  }) {

    try {

      return await VehicleProvider

        .findVehicle({

          make,

          model,

          year,

          vehicleType

        })

    }

    catch (error) {

      console.warn(

        '[VehicleEngine] Vehicle provider failed:',

        error

      )


      return null

    }

  }


  // ====================================================
  // SEARCH
  // ====================================================

  static async search({

    vehicleType,

    make,

    model,

    year,

    products = []

  }) {

    const safeProducts =

      Array.isArray(products)

        ? products

        : []


    // --------------------------------------------------
    // VEHICLE
    // --------------------------------------------------

    let vehicle = null


    try {

      vehicle =

        await this.findVehicle({

          make,

          model,

          year,

          vehicleType

        })

    }

    catch (error) {

      console.warn(

        '[VehicleEngine] Vehicle lookup failed:',

        error

      )

    }


    // --------------------------------------------------
    // OEM SPECIFICATIONS
    // --------------------------------------------------

    let oem = null


    try {

      oem =

        await OEMCompatibilityEngine

          .search({

            make,

            model,

            year,

            vehicleType

          })

    }

    catch (error) {

      console.warn(

        '[VehicleEngine] OEM lookup failed. Continuing with explicit compatibility.',

        error

      )

    }


    // --------------------------------------------------
    // TECHNICAL COMPATIBILITY
    // --------------------------------------------------

    const matched =

      safeProducts.filter(

        product =>

          productMatchesVehicle({

            product,

            make,

            model,

            year,

            vehicleType,

            oem

          })

      )


    // --------------------------------------------------
    // TYPE RESULTS
    // --------------------------------------------------

    const tires =

      matched.filter(

        product =>

          normalizeProductType(
            product
          ) === 'tire'

      )


    const batteries =

      matched.filter(

        product =>

          normalizeProductType(
            product
          ) === 'battery'

      )


    const oils =

      matched.filter(

        product =>

          normalizeProductType(
            product
          ) === 'oil'

      )


    // --------------------------------------------------
    // RETURN
    // --------------------------------------------------

    return {

      vehicle,

      oem,

      tires,

      batteries,

      oils,

      products:
        matched

    }

  }

  // FILTER TIRES
  // ====================================================

  static filterTires({

    products = [],

    vehicleType,

    make,

    model,

    year,

    oem

  }) {

    const safeProducts =

      Array.isArray(products)

        ? products

        : []


    return safeProducts.filter(

      product => {

        if (

          normalizeProductType(
            product
          ) !== 'tire'

        ) {

          return false

        }


        return productMatchesVehicle({

          product,

          make,

          model,

          year,

          vehicleType,

          oem

        })

      }

    )

  }


  // ====================================================
  // FILTER BATTERIES
  // ====================================================

  static filterBatteries({

    products = [],

    vehicleType,

    make,

    model,

    year,

    oem

  }) {

    const safeProducts =

      Array.isArray(products)

        ? products

        : []


    return safeProducts.filter(

      product => {

        if (

          normalizeProductType(
            product
          ) !== 'battery'

        ) {

          return false

        }


        return productMatchesVehicle({

          product,

          make,

          model,

          year,

          vehicleType,

          oem

        })

      }

    )

  }


  // ====================================================
  // FILTER OILS
  // ====================================================

  static filterOils({

    products = [],

    vehicleType,

    make,

    model,

    year,

    oem

  }) {

    const safeProducts =

      Array.isArray(products)

        ? products

        : []


    return safeProducts.filter(

      product => {

        if (

          normalizeProductType(
            product
          ) !== 'oil'

        ) {

          return false

        }


        return productMatchesVehicle({

          product,

          make,

          model,

          year,

          vehicleType,

          oem

        })

  // ====================================================

  static filterAll(

      ...this.filterTires(
        params
      ),

      ...this.filterBatteries(
        params
      ),

      ...this.filterOils(
        params
      )

    ]

  }

}

// ======================================================
// COLLECT VALUES
// ======================================================

const collectValues = value => {

  const values = []


  const add = item => {

    if (
      item === null ||
      item === undefined ||
      item === ''
    ) {

      return

    }


    if (
      Array.isArray(item)
    ) {

      item.forEach(add)

      return

    }


    if (
      typeof item === 'object'
    ) {

      Object.values(item)
        .forEach(add)

      return

    }


    values.push(item)

  }


  add(value)


  return values

}

const normalizeText = value => {

  return String(value ?? '')
    .trim()
    .toLowerCase()
    .replace(/ط£|ط¥|ط¢/g, 'ط§')
    .replace(/ط©/g, 'ظ‡')
    .replace(/ظ‰/g, 'ظٹ')
    .replace(/\s+/g, ' ')

}


// ======================================================
// UNIQUE VALUES
// ======================================================

const uniqueValues = values => {

  const seen =
    new Set()


  const result = []


  collectValues(values)
    .forEach(value => {

      const text =
        String(value ?? '')
          .trim()


      if (!text) {

        return

      }


      const key =
        normalizeText(text)


      if (
        !key ||
        seen.has(key)
      ) {

        return

      }


      seen.add(key)

      result.push(text)

    })


  return result

}
